8.Similarity vs accuracy

Prepare data

rm(list=setdiff(ls(),c("val_tcr_remain_list","val_cdr3_similar_list","unseen_pep_list","uniq_val_pep","pep_dist","uniq_train_pep","uniq_train_mhc")))
pep_dist_df<-data.frame(peptide=uniq_val_pep,mindis=apply(pep_dist, 1, min)) 

result<-read.table("ensembl_NoRand_feb20_CPU.txt",header = T,sep=",")
result$tcr<-paste(result$vaseq,result$cdr3a,result$vbseq,result$cdr3b,sep='_')
result$cdr3ab<-paste(result$cdr3a,result$cdr3b,sep='_')
result<-merge(result,pep_dist_df,by="peptide",all.x = T)

Plot results

plot(0, 0, type = "n", xlim = c(-0.1, 5), ylim = c(0.6, 1), 
     cex.lab = 1.2,
     cex.axis = 1.2,
     xlab = "Min. peptide distance between validation and training", ylab = "AUROC")

for (i in c(0,1,2,3,4,5)) {
  result_filtered<-result[result$mindis>=i,]
  auc<-PRROC::roc.curve((1-result_filtered$avg_rank[result_filtered$label==1]),(1-result_filtered$avg_rank[result_filtered$label==0]))
  points(jitter(i, amount = 0.1), auc$auc, col = "blue", pch = 19)
}
